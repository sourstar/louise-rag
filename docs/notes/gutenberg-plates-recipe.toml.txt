# THE CLEANING CHANGED, so its version moves (invariant 3: "any change to
# chunking or cleaning bumps the version of the thing that changed,
# explicitly"). `pg_epub.read_clean` reflows an epub's paragraphs;
# `pg_text.read_clean` preserves the .txt's hard wrap at ~70 characters.
# Measured on the node 2026-09-11: 0 of 60 books holding both forms
# produced an identical content_hash, and 6 of 1,228 chunks hashed the
# same — 0.5%.
#
# `cleaning_version` is inside `space_id`, which is the point. Leaving it
# at 1 would put this collection in the SAME space as the plain-text one
# while the two produce different text for the same book, and a vector
# admitted across that boundary would be a vector of text the record does
# not contain. Nothing would detect it: both halves compute their ids
# honestly. One integer prevents it.
#
# AT THE TOP, and deliberately: in TOML a bare key belongs to the table
# above it, so these written after a [table] header would land inside it.
# `hold collection create` refuses that rather than accepting a layer with
# a `cleaning_version` in it — which is how this file was caught once.

# THE LAYER, DECLARED EXPLICITLY — preflight item 2, and the reason it is
# item 2. The gutenberg plugin declares `chunking_profile = "book"`, and
# `book` is NOT one of 06 §6.3's four chunkers (`fixed`, `cdc`,
# `abstract`, `verbatim`); it is a grandfathered name kept so nothing
# already pinned moves. A recipe that omits `layers` inherits it — and
# 03 §3.5 gates T1's overlap-band check on `fixed` layers, so a Gutenberg
# collection whose layer says `book` is NEVER T1-checked by anyone, for
# the life of the collection. `layers` is inside `collection_id`, so that
# cannot be corrected later; it can only be replaced.
#
# `fixed{tokens: 1000, overlap: 120}` at version 1 cuts BYTE-IDENTICALLY
# to `book` at this node's live settings (CHUNK_TOKENS=1000,
# CHUNK_OVERLAP=120, CHUNK_VERSION=1, read from the worker 2026-09-10)
# and produces the same chunk ids — so declaring it costs nothing and
# buys T1 forever.
[[layers]]
rule = "default"

[layers.chunker]
id = "fixed"
version = 1
params = { tokens = 1000, overlap = 120 }
