#
#   [space.encoder]
#   weights = "sha256:…"
#   label   = "…"
#   [space.encoder.engine]
#   name = "llama.cpp"   build = "b10630"   variant = "sycl"
#   image  = "sha256:<the published RAGE image that ran>"
#   args   = [ … the model's cmd from rage.toml, ${WEIGHTS} and ${PORT} as written … ]
#   device = "Intel(R) Arc(TM) Pro B70 Graphics"
#
# and a symmetric model (one encoder for documents and queries, as here) has
# NO [space.query.encoder] at all: naming the same weights there makes the
# space asymmetric on paper, and version 3 would then pin and gate a second,
# identical reference (09 §9.12, query_vectors). Migration-plan step 13a.

# THE LAYER, DECLARED EXPLICITLY — preflight item 2, and the reason it is
# item 2. The gutenberg plugin declares `chunking_profile = "book"`, and
# `book` is NOT one of 06 §6.3's four chunkers (`fixed`, `cdc`,
# `abstract`, `verbatim`); it is a grandfathered name kept so nothing
# already pinned moves. A recipe that omits `layers` inherits it — and
# 03 §3.5 gates T1's overlap-band check on `fixed` layers, so a Gutenberg
# collection whose layer says `book` is NEVER T1-checked by anyone, for
# the life of the collection. `layers` is inside `collection_id`, so that
# cannot be corrected later; it can only be replaced.
#
# `fixed{tokens: 1000, overlap: 120}` at version 1 cuts BYTE-IDENTICALLY
# to `book` at this node's live settings (CHUNK_TOKENS=1000,
# CHUNK_OVERLAP=120, CHUNK_VERSION=1, read from the worker 2026-09-10)
# and produces the same chunk ids — so declaring it costs nothing and
# buys T1 forever.
[[layers]]
rule = "default"

[layers.chunker]
id = "fixed"
version = 1
params = { tokens = 1000, overlap = 120 }
